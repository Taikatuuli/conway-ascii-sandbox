# Karaoke Language Learning App — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a single-file browser app that embeds a YouTube video alongside Claude-translated, phonetically-annotated lyrics that highlight in sync with playback.

**Architecture:** All HTML, CSS, and JS live in `karaoke.html`. Three screens (input, loading, player) are toggled via CSS class swaps. State lives in a single `APP` object. The YouTube iframe API drives playback; a 500ms polling loop drives lyric highlighting.

**Tech Stack:** Vanilla HTML/CSS/JS, YouTube iframe API (CDN), Anthropic Messages API (browser fetch with `x-api-key` header)

```js
// ── UTILITIES ─────────────────────────────────────────────────────────────────
function extractVideoId(url) {
  const m = url.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/|youtube\.com\/embed\/)([^&\n?#]{11})/);
  return m ? m[1] : null;
}

function showError(id, msg) {
  const el = document.getElementById(id);
  el.textContent = msg;
  el.style.display = 'block';
}
function clearError(id) { document.getElementById(id).style.display = 'none'; }

// ── API KEY GATE ──────────────────────────────────────────────────────────────
function initApiKey() {
  APP.apiKey = localStorage.getItem('anthropic_api_key');
  if (!APP.apiKey) {
    document.getElementById('apikey-prompt').style.display = 'flex';
    document.getElementById('apikey-prompt').style.flexDirection = 'column';
    document.getElementById('apikey-prompt').style.gap = '6px';
  }
}

function getOrSaveApiKey() {
  if (APP.apiKey) return APP.apiKey;
  const val = document.getElementById('input-apikey').value.trim();
  if (!val) {
    showError('apikey-error', 'Please enter your Anthropic API key.');
    return null;
  }
  localStorage.setItem('anthropic_api_key', val);
  APP.apiKey = val;
  return val;
}

// ── START BUTTON ──────────────────────────────────────────────────────────────
document.getElementById('btn-start').addEventListener('click', async () => {
  clearError('input-error');
  clearError('apikey-error');

  const key = getOrSaveApiKey();
  if (!key) return;

  const url = document.getElementById('input-url').value.trim();
  const language = document.getElementById('input-language').value;
  const rawLyrics = document.getElementById('input-lyrics').value;

  if (!extractVideoId(url)) {
    showError('input-error', 'Could not find a YouTube video ID in that URL.');
    return;
  }
  if (!language) {
    showError('input-error', 'Please select a target language.');
    return;
  }
  const lines = rawLyrics.split('\n').map(l => l.trim()).filter(Boolean);
  if (lines.length < 2) {
    showError('input-error', 'Please paste at least 2 lyric lines.');
    return;
  }

  APP.language = language;
  await startTranslation(lines, url);
});

// Run on load
initApiKey();
```

```js
// ── TRANSLATION PIPELINE ──────────────────────────────────────────────────────
async function startTranslation(lines, youtubeUrl) {
  showScreen('screen-loading');
  document.getElementById('loading-msg').textContent = `Translating ${lines.length} lines…`;
  document.getElementById('loading-sub').textContent = 'This takes a few seconds';

  document.getElementById('btn-cancel').onclick = () => {
    APP.apiKey = null; // don't clear localStorage — just reset session
    showScreen('screen-input');
  };

  let lyrics;
  try {
    lyrics = await translateLyrics(lines, APP.language, APP.apiKey);
  } catch (err) {
    showScreen('screen-input');
    if (err.status === 401) {
      localStorage.removeItem('anthropic_api_key');
      APP.apiKey = null;
      document.getElementById('apikey-prompt').style.display = 'flex';
      showError('apikey-error', 'Invalid API key. Please re-enter it.');
    } else {
      showError('input-error', `Translation failed: ${err.message}`);
    }
    return;
  }

  APP.lyrics = lyrics;
  initPlayer(extractVideoId(youtubeUrl));
}

async function translateLyrics(lines, language, apiKey) {
  const prompt = `You are a language tutor. The user is learning ${language}. Translate the following ${language} song lyrics and provide phonetic guides and key vocabulary.

Return ONLY a JSON array with exactly ${lines.length} objects — one per input line — in this exact shape:
[{"original":"...","translation":"...","phonetic":"...","keywords":[{"word":"...","meaning":"..."}]}]

Rules:
- "phonetic": natural romanization for the language (pinyin for Mandarin, romaji for Japanese, Revised Romanization for Korean, loose phonetics for others)
- "keywords": 3–6 notable vocabulary words per line, chosen for learning value
- Return ONLY the raw JSON array. No markdown fences, no explanation text.

Lyrics (${lines.length} lines):
${lines.map((l, i) => `${i + 1}. ${l}`).join('\n')}`;

  const res = await fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: {
      'x-api-key': apiKey,
      'anthropic-version': '2023-06-01',
      'content-type': 'application/json',
    },
    body: JSON.stringify({
      model: 'claude-sonnet-4-6',
      max_tokens: 4096,
      messages: [{ role: 'user', content: prompt }],
    }),
  });

  if (!res.ok) {
    const err = new Error(`API error ${res.status}`);
    err.status = res.status;
    throw err;
  }

  const data = await res.json();
  const text = data.content[0].text.trim();

  let parsed;
  try {
    parsed = JSON.parse(text);
  } catch {
    // Claude returned something wrapped — try to extract JSON array
    const match = text.match(/\[[\s\S]*\]/);
    if (!match) throw new Error('Could not parse translation response as JSON.');
    parsed = JSON.parse(match[0]);
  }

  // Ensure every line has an entry; fall back gracefully for missing ones
  return lines.map((original, i) => parsed[i] ?? { original, translation: '', phonetic: '', keywords: [] });
}
```

```js
// ── YOUTUBE PLAYER ────────────────────────────────────────────────────────────
let ytApiReady = false;
let pendingVideoId = null;

window.onYouTubeIframeAPIReady = function() {
  ytApiReady = true;
  if (pendingVideoId) createPlayer(pendingVideoId);
};

function initPlayer(videoId) {
  showScreen('screen-player');
  renderLyrics(); // Task 5 — will be a no-op until implemented
  if (ytApiReady) {
    createPlayer(videoId);
  } else {
    pendingVideoId = videoId;
  }
}

function createPlayer(videoId) {
  pendingVideoId = null;
  APP.player = new YT.Player('yt-player', {
    videoId,
    playerVars: { autoplay: 1, rel: 0, modestbranding: 1 },
    events: {
      onReady: onPlayerReady,
      onStateChange: onPlayerStateChange,
    },
  });
}

function onPlayerReady() {
  waitForDuration();
}

function waitForDuration() {
  const dur = APP.player.getDuration();
  if (dur > 0) {
    assignTimestamps(dur);
    startPolling();
    return;
  }
  if (APP.durationRetries++ < 10) {
    setTimeout(waitForDuration, 500);
  } else {
    // Give up — use line count as fallback (assume 3 min video)
    assignTimestamps(180);
    startPolling();
  }
}

function onPlayerStateChange(e) {
  if (e.data === YT.PlayerState.ENDED) showRecap();
}
```

```js
function assignTimestamps(duration) {
  const count = APP.lyrics.length;
  APP.timestamps = APP.lyrics.map((_, i) => (i / count) * duration);
}

function currentLineFor(currentTime) {
  let idx = 0;
  for (let i = 0; i < APP.timestamps.length; i++) {
    if (APP.timestamps[i] <= currentTime) idx = i;
    else break;
  }
  return idx;
}

function startPolling() {
  if (APP.pollInterval) clearInterval(APP.pollInterval);
  APP.pollInterval = setInterval(() => {
    if (!APP.player || typeof APP.player.getCurrentTime !== 'function') return;
    const t = APP.player.getCurrentTime();
    const next = currentLineFor(t);
    if (next !== APP.currentLine) {
      updateCurrentLine(next);
    }
  }, 500);
}

function updateCurrentLine(index) {
  APP.currentLine = index;
  const panel = document.getElementById('lyrics-panel');
  const lines = panel.querySelectorAll('.lyric-line');

  lines.forEach((el, i) => {
    el.classList.remove('current', 'past', 'upcoming');
    if (i < index) el.classList.add('past');
    else if (i === index) el.classList.add('current');
    else el.classList.add('upcoming');
  });

  // Scroll current line into center of panel
  const currentEl = lines[index];
  if (currentEl) {
    currentEl.scrollIntoView({ behavior: 'smooth', block: 'center' });
  }
}
```

```js
// ── TAP-TO-TRANSLATE ──────────────────────────────────────────────────────────
let tooltipTimer = null;

document.getElementById('lyrics-panel').addEventListener('click', e => {
  const wordEl = e.target.closest('.word');
  if (!wordEl) return;

  const lineIndex = parseInt(wordEl.dataset.line, 10);
  if (lineIndex !== APP.currentLine) return; // only current line

  const word = wordEl.textContent.trim().replace(/[.,!?;:'"…—]/g, '');
  const lineData = APP.lyrics[lineIndex];
  const kw = lineData?.keywords.find(
    k => k.word.toLowerCase() === word.toLowerCase()
  );

  const tooltip = document.getElementById('tooltip');
  document.getElementById('tooltip-word').textContent = word;
  document.getElementById('tooltip-meaning').textContent =
    kw ? kw.meaning : 'No translation available';

  // Position tooltip near word
  const rect = wordEl.getBoundingClientRect();
  const tt = tooltip;
  tt.style.display = 'block';
  const ttW = 240;
  let left = rect.left;
  if (left + ttW > window.innerWidth - 8) left = window.innerWidth - ttW - 8;
  tt.style.left = `${left}px`;
  tt.style.top = `${rect.bottom + 6}px`;

  // Auto-dismiss after 3s
  clearTimeout(tooltipTimer);
  tooltipTimer = setTimeout(hideTooltip, 3000);
});

document.addEventListener('mousedown', e => {
  if (!e.target.closest('#tooltip') && !e.target.closest('.word')) hideTooltip();
});

function hideTooltip() {
  clearTimeout(tooltipTimer);
  document.getElementById('tooltip').style.display = 'none';
}
```

```js
function showRecap() {
  const allKeywords = uniqueKeywords(APP.lyrics);

  document.getElementById('recap-title').innerHTML =
    `You learned <span>${allKeywords.length} words</span> in ${APP.language}!`;

  const grid = document.getElementById('recap-grid');
  grid.innerHTML = '';
  allKeywords.forEach(kw => {
    const card = document.createElement('div');
    card.className = 'recap-kw';
    card.innerHTML = `<div class="recap-kw-word">${kw.word}</div><div class="recap-kw-meaning">${kw.meaning}</div>`;
    grid.appendChild(card);
  });

  document.getElementById('modal-recap').style.display = 'flex';
}

function uniqueKeywords(lyrics) {
  const seen = new Set();
  const result = [];
  for (const line of lyrics) {
    for (const kw of line.keywords) {
      const key = kw.word.toLowerCase();
      if (!seen.has(key)) {
        seen.add(key);
        result.push(kw);
      }
    }
  }
  return result;
}

  APP.currentLine = -1;

});
```
